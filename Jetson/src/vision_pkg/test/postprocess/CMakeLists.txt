# CPU production postprocessing; no ROS, TensorRT, CUDA or model assets.
cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vision_postprocess_tests LANGUAGES CXX)
  enable_testing()
  if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
  endif()
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
get_filename_component(VISION_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
if(NOT TARGET vision_postprocess)
  add_library(vision_postprocess STATIC "${VISION_ROOT}/yolo/postprocess.cpp")
  target_include_directories(vision_postprocess PUBLIC "${VISION_ROOT}" ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(vision_postprocess PUBLIC ${OpenCV_LIBS})
  target_compile_features(vision_postprocess PUBLIC cxx_std_17)
endif()
add_executable(test_postprocess test_postprocess.cpp)
target_link_libraries(test_postprocess PRIVATE vision_postprocess)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(test_postprocess PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(COMMAND ament_add_test)
  ament_add_test(test_postprocess COMMAND $<TARGET_FILE:test_postprocess>
    GENERATE_RESULT_FOR_RETURN_CODE_ZERO TIMEOUT 120)
else()
  add_test(NAME test_postprocess COMMAND test_postprocess)
  set_tests_properties(test_postprocess PROPERTIES TIMEOUT 120)
endif()
