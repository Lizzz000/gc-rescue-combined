# 只下载两个固定版本的头文件，校验哈希；依赖不进入源码或板端视频目录。
function(record_header name url hash)
  set(path "${RECORD_DEPS_DIR}/${name}")
  if(NOT EXISTS "${path}")
    get_filename_component(parent "${path}" DIRECTORY)
    file(MAKE_DIRECTORY "${parent}")
    file(DOWNLOAD "${url}" "${path}" TLS_VERIFY ON EXPECTED_HASH "SHA256=${hash}" STATUS result)
    list(GET result 0 code)
    if(NOT code EQUAL 0)
      file(REMOVE "${path}")
      message(FATAL_ERROR "Cannot download ${name}: ${result}. Populate RECORD_DEPS_DIR from the documented pinned URL.")
    endif()
  endif()
  file(SHA256 "${path}" actual)
  if(NOT actual STREQUAL hash)
    message(FATAL_ERROR "Pinned dependency hash mismatch: ${path}")
  endif()
endfunction()
