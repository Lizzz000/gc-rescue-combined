cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(record_control_tests LANGUAGES CXX)
  enable_testing()
endif()
add_executable(test_record_control test_control_requests.cpp)
add_executable(test_record_protocol test_stream_protocol.cpp ../../src/stream_protocol.cpp)
foreach(target test_record_control test_record_protocol)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE ../../src)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
add_test(NAME record_control_requests COMMAND test_record_control)
add_test(NAME record_stream_protocol COMMAND test_record_protocol)
