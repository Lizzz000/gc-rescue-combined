cmake_minimum_required(VERSION 3.16)
project(teleop_pkg VERSION 0.4.0 LANGUAGES CXX)

# clangd复用本包真实编译命令；仅生成开发元数据，不改变编译选项。
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake QUIET)
option(TELEOP_USE_AMENT "Install as an ament package" ${ament_cmake_FOUND})
option(TELEOP_BUILD_PC "Build SDL2 keyboard/gamepad client" ${WIN32})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(serial_default ON)
else()
  set(serial_default OFF)
endif()
option(TELEOP_BUILD_SERIAL "Build Linux termios serial bridge" ${serial_default})
if(TELEOP_USE_AMENT)
  find_package(ament_cmake REQUIRED)
endif()

add_library(teleop_core STATIC src/control.cpp src/platform/platform.cpp)
target_include_directories(teleop_core PUBLIC include PRIVATE src)
if(MSVC)
  target_compile_options(teleop_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(teleop_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_link_libraries(teleop_core PUBLIC ws2_32)
endif()
if(TELEOP_USE_AMENT)
  set(teleop_bin lib/${PROJECT_NAME})
else()
  set(teleop_bin bin)
endif()
if(TELEOP_BUILD_SERIAL)
  add_executable(serial_control src/app/serial_control.cpp src/hidraw/beitong.cpp src/termios/serial_port.cpp)
  target_include_directories(serial_control PRIVATE src)
  target_link_libraries(serial_control PRIVATE teleop_core)
  install(TARGETS serial_control DESTINATION ${teleop_bin})
endif()
if(TELEOP_BUILD_PC)
  find_package(SDL2 2.0.14 REQUIRED CONFIG)
  add_executable(pc_control src/app/pc_control.cpp src/sdl/pc_input.cpp)
  target_include_directories(pc_control PRIVATE src)
  target_compile_definitions(pc_control PRIVATE SDL_MAIN_HANDLED)
  target_link_libraries(pc_control PRIVATE teleop_core SDL2::SDL2)
  install(TARGETS pc_control DESTINATION ${teleop_bin})
endif()
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
install(FILES config/70-teleop-beitong.rules DESTINATION share/${PROJECT_NAME}/config)
if(TELEOP_USE_AMENT)
  ament_package()
endif()
